markdown · docs/项目约束.md（节选）
# 项目约束（AI 编码时必须遵守）
1. 统一使用 MyBatis-Plus，禁止原生 JDBC
2. 所有新增接口必须加 @RequiresPermissions 权限注解
3. 数据库操作统一走 Service 层，Controller 禁止直接调 Mapper
4. 涉及多表写入的方法必须 @Transactional
5. 返回体统一使用 R 包装，禁止直接返回 Map
6. Listener 消费必须幂等（唯一键或状态位）
7. Job 必须防重入（分布式锁或状态位）
8. 版本：Spring Boot 3.2.x / MyBatis-Plus 3.5.x / MySQL 8.0
🎯 要点：约束必须"能检查对错"，禁止"尽量/建议"。这是 3.x 方案和 5.1 编码的对照基准。

package com.example.order.controller;

import com.example.order.domain.dto.OrderDTO;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.autoconfigure.web.servlet.AutoConfigureMockMvc;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.http.MediaType;
import org.springframework.test.web.servlet.MockMvc;

import java.math.BigDecimal;
import java.util.List;

import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.post;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.jsonPath;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

@SpringBootTest
@AutoConfigureMockMvc
class OrderControllerTest {

    @Autowired private MockMvc mockMvc;
    @Autowired private ObjectMapper objectMapper;

    /** 场景 #1：合法入参 → 200 + data.orderId 存在 */
    @Test
    void 创建订单_合法入参_返回订单号() throws Exception {
        OrderDTO dto = new OrderDTO();
        dto.setUserId(88L);
        dto.setItemIds(List.of(1L, 2L));
        dto.setAmount(new BigDecimal("299.00"));

        mockMvc.perform(post("/order")
                .contentType(MediaType.APPLICATION_JSON)
                .content(objectMapper.writeValueAsString(dto)))
            .andExpect(status().isOk())
            .andExpect(jsonPath("$.code").value(200))
            .andExpect(jsonPath("$.data.orderId").isNumber());
    }

    /** 场景 #2：商品不存在 → 400 + 提示 */
    @Test
    void 创建订单_商品不存在_返回400() throws Exception {
        OrderDTO dto = new OrderDTO();
        dto.setUserId(88L);
        dto.setItemIds(List.of(99999L));
        dto.setAmount(new BigDecimal("100.00"));

        mockMvc.perform(post("/order")
                .contentType(MediaType.APPLICATION_JSON)
                .content(objectMapper.writeValueAsString(dto)))
            .andExpect(status().isBadRequest())
            .andExpect(jsonPath("$.code").value(400))
            .andExpect(jsonPath("$.msg").value("商品不存在"));
    }

    /** 场景 #4：未授权 → 403 */
    @Test
    void 创建订单_未授权_返回403() throws Exception {
        mockMvc.perform(post("/order").contentType(MediaType.APPLICATION_JSON)
                .content("{}"))
            .andExpect(status().isForbidden());
    }
}

package com.example.order.controller;

import com.example.common.core.domain.R;
import com.example.order.domain.dto.OrderDTO;
import com.example.order.domain.vo.OrderVO;
import com.example.order.service.OrderService;
import lombok.RequiredArgsConstructor;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.*;
import org.apache.shiro.authz.annotation.RequiresPermissions;

import jakarta.validation.Valid;

/**
 * 订单创建接口
 */
@RestController
@RequestMapping("/order")
@RequiredArgsConstructor
public class OrderController {

    private final OrderService orderService;

    /**
     * 创建订单
     *
     * @param dto 订单信息
     * @return 创建成功返回订单号
     */
    @PostMapping
    @RequiresPermissions("order:order:add")
    public R<OrderVO> create(@Valid @RequestBody OrderDTO dto) {
        return R.ok(orderService.createOrder(dto));
    }
}

Service 实现（核心逻辑：金额校验 + 库存扣减 + 状态流转）

java · OrderServiceImpl.java
package com.example.order.service.impl;

import com.example.common.exception.ServiceException;
import com.example.order.domain.Order;
import com.example.order.domain.dto.OrderDTO;
import com.example.order.domain.vo.OrderVO;
import com.example.order.mapper.OrderMapper;
import com.example.order.service.OrderService;
import com.example.stock.service.StockService;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Slf4j
@Service
@RequiredArgsConstructor
public class OrderServiceImpl implements OrderService {

    private final OrderMapper orderMapper;
    private final StockService stockService;

    @Override
    @Transactional // 约束 #4：多表写入必须事务
    public OrderVO createOrder(OrderDTO dto) {
        // 1. 金额校验（验收场景 #3）
        if (dto.getAmount().signum() <= 0) {
            throw new ServiceException("金额必须大于0");
        }
        // 2. 库存扣减（依赖澄清 #3：失败回滚）
        stockService.deduct(dto.getItemIds());
        // 3. 落库
        Order order = new Order();
        order.setUserId(dto.getUserId());
        order.setAmount(dto.getAmount());
        order.setStatus("PENDING");
        orderMapper.insert(order);
        log.info("订单创建成功 orderId={}", order.getOrderId());
        // 4. 返回 VO
        OrderVO vo = new OrderVO();
        vo.setOrderId(order.getOrderId());
        vo.setStatus(order.getStatus());
        return vo;
    }
}

防线为什么这套流程"不跑偏"

防线作用对应环节
类型化模板每种产物格式固定，AI 没有发挥空间3.1 / 3.2 / 3.3
验收场景前置方案阶段写死"怎么算对"3.x 方案 → 4.2 测试
澄清闸门歧义不进方案1.2 /clarify
约束硬规则能检查对错的规则，AI 逐条对照2.1 /constraints
角色分离人写测试 ≠ AI 实现，防"骗绿"4.2 ↔ 5.1
收敛兜底实现后逐条核对，防规格漂移5.2 /accept
三个最常见的坑
1️⃣ 跳过 1.2 澄清直接写方案 → 歧义带进编码
2️⃣ 让实现 AI 顺便写测试 → 它会改弱断言"骗绿"
3️⃣ 约束写成"尽量/建议" → 无法检查 = 无效约束
